// Put these inside match /databases/{database}/documents {

function signedIn() {
  return request.auth != null;
}

function isAdmin() {
  return signedIn() &&
      exists(/databases/$(database)/documents/admins/$(request.auth.uid));
}

match /admins/{adminId} {
  allow get: if signedIn() && request.auth.uid == adminId;
  allow list: if isAdmin();
  allow create, update, delete: if isAdmin();
}

match /travelers/{travelerId} {
  allow read: if signedIn() &&
      (request.auth.uid == travelerId || isAdmin());

  allow create: if signedIn() &&
      request.auth.uid == travelerId &&
      request.resource.data.role == 'traveler';

  allow update, delete: if signedIn() &&
      (request.auth.uid == travelerId || isAdmin());
}

match /vendors/{vendorId} {
  allow read: if signedIn();

  allow create: if signedIn() &&
      request.auth.uid == vendorId &&
      request.resource.data.role == 'vendor';

  allow update, delete: if signedIn() &&
      (request.auth.uid == vendorId || isAdmin());
}
